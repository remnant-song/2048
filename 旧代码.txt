 从二维vector的空位置中随机返回一个坐标（行, 列）
 若没有空位置，返回(-1, -1)
pair<int, int> getRandomEmptyPos(const vector<vector<char>>& board) {
	vector<pair<int, int>> emptyPositions;
	for (int i = 1; i <= 9; i += 2) {
		for (int j = 2; j <= 18; j += 4) {
			if (board[i][j] == ' ') {
				emptyPositions.emplace_back(i, j);
			}
		}
	}
	if (emptyPositions.empty()) {
		return { -1, -1 };
	}
	 优化随机数：改用mt19937避免秒级重复
	unsigned seed = chrono::system_clock::now().time_since_epoch().count();
	mt19937 rng(seed);
	uniform_int_distribution<int> dist(0, emptyPositions.size() - 1);
	int randomIndex = dist(rng);
	return emptyPositions[randomIndex];
}

 辅助函数：获取5×5区域的原始行列索引（将1,3,5,7,9映射为0~4；2,6,10,14,18映射为0~4）
int getRowIndex(int i) { return (i - 1) / 2; }  i=1→0, i=3→1, ..., i=9→4
int getColIndex(int j) { return (j - 2) / 4; }  j=2→0, j=6→1, ..., j=18→4

 辅助函数：通过5×5区域的索引获取实际board的坐标
int getActualRow(int row) { return 1 + row * 2; }  row=0→1, row=1→3, ..., row=4→9
int getActualCol(int col) { return 2 + col * 4; }  col=0→2, col=1→6, ..., col=4→18
void addAndCLose()
{
	for (int i = 1; i <= 9; i += 2)普通更新：向左相加且靠拢
	{
		char a = ' ';用来暂存待加数
		for (int j = 18; j >= 2; j -= 4)行内循环
		{
			if (board[i][j] != ' ')当前的位置不为空
			{
				if (a != ' ')前面存在待加数
				{
					board[i][j] = ((a - '0') + (board[i][j] - '0') + '0');当前位置和前一个相加
					a = ' ';待加数变量清空
				}
				else待加数为空，给变量赋值
				{
					a = board[i][j];
					board[i][j] = ' ';
				}
			}
		}
		board[i][2] = a;奇数个有效数的话，会有一个剩余的数被a携带，原位置被清空。将这个剩余的数赋给最后一个位置。

		int cursor = 2;
		for (int j = 2; j <= 18; j += 4)从左至右依次遍历，将数据向左靠拢
		{
			if (board[i][j] != ' ')
			{
				board[i][cursor] = board[i][j];将该数放到游标所指位置
				cursor += 4;游标向后挪动
				board[i][j] = ' ';当前位置置空
			}
		}
	}
}
void updateBoard(char forward)
{
	if (forward == 'd')向右，需进行左右反转
	{
		cout << "d,左右反转：" << endl;
		for (int i = 1; i <= 9; i += 2)
		{
			for (int j = 2; j <=18; j += 4)
			{
				swap(board[i][j], board[i][20-j]);
			}
		}
		addAndCLose();
		恢复反转
		for (int i = 1; i <= 9; i += 2) {
			for (int j = 2; j <= 18; j += 4) {
				swap(board[i][j], board[i][20 - j]);
			}
		}
	}
	else if (forward == 'w')向上，需要逆时针90度
	{
		 先复制原始的值
		vector<vector<char>> temp(5, vector<char>(5));
		for (int i = 1; i <= 9; i += 2) {
			for (int j = 2; j <= 18; j += 4) {
				int row = getRowIndex(i);
				int col = getColIndex(j);
				temp[row][col] = board[i][j];
			}
		}

		 逆时针90度旋转
		for (int row = 0; row < 5; ++row) {
			for (int col = 0; col < 5; ++col) {
				int newRow = 4 - col;
				int newCol = row;
				 映射到实际board的坐标并赋值
				int actualI = getActualRow(newRow);
				int actualJ = getActualCol(newCol);
				board[actualI][actualJ] = temp[row][col];
			}
		}
		addAndCLose();
		恢复旋转
		vector<vector<char>> temp2(5, vector<char>(5));
		for (int i = 1; i <= 9; i += 2) {
			for (int j = 2; j <= 18; j += 4) {
				int row = getRowIndex(i);
				int col = getColIndex(j);
				temp2[row][col] = board[i][j];
			}
		}
		 执行顺时针90度旋转（逆变换）
		for (int row = 0; row < 5; ++row) {
			for (int col = 0; col < 5; ++col) {
				int newRow = 4 - col;
				int newCol = row;
				int actualI = getActualRow(newRow);
				int actualJ = getActualCol(newCol);
				board[actualI][actualJ] = temp2[col][4 - row];  顺时针旋转公式
			}
		}
	}
	else if (forward == 's')向下，需要顺时针90度
	{
		 先复制原始的值
		vector<vector<char>> temp(5, vector<char>(5));
		for (int i = 1; i <= 9; i += 2) {
			for (int j = 2; j <= 18; j += 4) {
				int row = getRowIndex(i);
				int col = getColIndex(j);
				temp[row][col] = board[i][j];
			}
		}

		 顺时针90度旋转
		for (int row = 0; row < 5; ++row) {
			for (int col = 0; col < 5; ++col) {
				int newRow = col;
				int newCol = 4 - row;
				 映射到实际board的坐标并赋值
				int actualI = getActualRow(newRow);
				int actualJ = getActualCol(newCol);
				board[actualI][actualJ] = temp[row][col];
			}
		}
		addAndCLose();
		恢复旋转
		vector<vector<char>> temp2(5, vector<char>(5));
		for (int i = 1; i <= 9; i += 2) {
			for (int j = 2; j <= 18; j += 4) {
				int row = getRowIndex(i);
				int col = getColIndex(j);
				temp2[row][col] = board[i][j];
			}
		}
		 执行逆时针90度旋转（逆变换）
		for (int row = 0; row < 5; ++row) {
			for (int col = 0; col < 5; ++col) {
				int newRow = 4 - col;
				int newCol = row;
				int actualI = getActualRow(newRow);
				int actualJ = getActualCol(newCol);
				board[actualI][actualJ] = temp2[newCol][4 - newRow];  逆时针旋转公式
			}
		}
	}
}
